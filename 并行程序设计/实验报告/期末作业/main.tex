	\section{引言}
	
	随着大数据时代的到来和计算需求的爆炸式增长，并行计算已成为现代高性能计算的核心技术。本学期的并行程序设计课程深入探讨了多种并行计算架构和编程模型，包括SIMD（单指令多数据）、Pthread\&OpenMP多线程并行、MPI多进程并行、GPU加速等。通过一系列递进式的实验，我们系统学习了各种并行编程技术，并以数论变换（NTT）多项式乘法作为核心算法载体，深入研究了不同并行优化策略的性能特征。
	
	数论变换作为快速傅里叶变换（FFT）在有限域上的对应算法，在密码学、信号处理、同态加密等领域有着广泛应用。其$O(n\log n)$的时间复杂度相比朴素$O(n^2)$算法具有巨大优势，但实际应用中仍需要通过并行优化来满足大规模计算需求。本期末报告将：
	
	\begin{enumerate}
		\item 系统总结本学期各项并行编程实验的核心成果和技术要点
		\item 展示综合多种并行技术的创新设计方案及其性能表现
		\item 深入分析不同并行策略的适用场景和优化效果
		\item 总结学习收获并展望未来发展方向
	\end{enumerate}
	
	\section{前期实验总结与回顾}
	
	本学期通过一系列循序渐进的实验，我们系统掌握了主流的并行编程技术。每个实验都针对NTT多项式乘法这一核心算法，探索了不同层次的并行优化策略。
	
	\subsection{SIMD向量化优化实验}
	
	\subsubsection{实验目标与内容}
	
	SIMD（Single Instruction Multiple Data）是现代处理器提供的数据级并行能力，通过单条指令同时处理多个数据元素来提升计算效率。本实验的主要目标包括：
	
	\begin{itemize}
		\item 理解SIMD指令集的工作原理和编程模型
		\item 掌握ARM NEON和x86 AVX指令集的使用方法
		\item 实现朴素多项式乘法和NTT算法的SIMD优化
		\item 引入Montgomery模约简算法进一步提升性能
	\end{itemize}

	\paragraph{Barrett规约在MPI中的应用}
	
	Barrett规约通过预计算避免了昂贵的除法运算：
	
	\begin{lstlisting}[language=C++, caption=Barrett规约结构]
		struct Barrett {
			unsigned long long mod, im;
			
			Barrett(unsigned long long _mod) : mod(_mod) {
				im = -1ULL / mod + 1;
			}
			
			unsigned long long multiply(unsigned long long a, unsigned long long b) const {
				unsigned long long z = a * b;
				unsigned long long x = (unsigned __int128)z * im >> 64;
				unsigned long long v = z - x * mod;
				if (v >= mod) v += mod;
				return v;
			}
		};
	\end{lstlisting}

	-
	
	\begin{lstlisting}[language=C++, caption=GPU蝶形运算核函数]
		__global__ void ntt_basic_kernel(LL *a, int len, int h, LL *w_table, LL p) {
			// blockIdx.x 标识当前是第几个蝶形运算组
			int j_base = blockIdx.x * h;
			// threadIdx.x 负责组内的计算
			for (int k = threadIdx.x; k < h / 2; k += blockDim.x) {
				int idx1 = j_base + k;
				int idx2 = j_base + k + h / 2;
				LL w = w_table[k];
				// 核心蝶形运算
				unsigned __int128 product = (unsigned __int128)a[idx2] * w;
				LL t = product % p;
				a[idx2] = (a[idx1] - t + p) % p;
				a[idx1] = (a[idx1] + t) % p;
			}
		}
	\end{lstlisting}
	
	\paragraph{Montgomery规约GPU实现}
	-
	
	\begin{lstlisting}[language=C++, caption=GPU Montgomery模乘]
		__device__ ULL montgomery_reduce(unsigned __int128 T, const MontgomeryParams* params) {
			ULL m = (ULL)T * params->n_prime;
			unsigned __int128 t = (T + (unsigned __int128)m * params->mod) >> 64;
			return (t >= params->mod) ? (t - params->mod) : t;
		}
	\end{lstlisting}

	\section{综合并行优化设计与创新}
	
	基于前期各项实验的经验积累，本次期末实验设计了三条综合并行优化路线，充分发挥不同并行技术的优势，实现了性能的进一步突破。
	
	\subsection{综合优化路线一：CPU多级优化路线}
	
	\subsubsection{设计思路}
	
	CPU优化路线采用渐进式优化策略，逐步引入更高级的优化技术：
	
	\begin{enumerate}
		\item \textbf{Level 1 - 基础NTT}：实现标准的迭代式NTT算法
		\item \textbf{Level 2 - Barrett优化}：引入Barrett规约减少模运算开销
		\item \textbf{Level 3 - Barrett + SIMD}：结合向量化指令加速计算
		\item \textbf{Level 4 - Barrett + SIMD + OpenMP}：利用多核并行和CRT
	\end{enumerate}
	
	\subsubsection{关键技术实现}
	
	\paragraph{Level 3: Barrett + SIMD优化}
	
	这一层级的优化结合了算法优化和硬件加速：
	
	\begin{lstlisting}[language=C++, caption=Barrett+SIMD核心实现]
		void NTT_Level3_Barrett_SIMD(const LL *a, const LL *b, LL *ab, int n, LL p) {
			// 使用对齐的内存以提高SIMD效率
			ULL *A = (ULL*)aligned_alloc(32, len * sizeof(ULL));
			ULL *B = (ULL*)aligned_alloc(32, len * sizeof(ULL));
			
			// 预计算所有旋转因子
			std::vector<std::vector<ULL>> w_tables;
			for (int h = 2; h <= len; h <<= 1) {
				std::vector<ULL> w_table(h/2);
				ULL wn = barrett.power(3, (barrett.mod - 1) / h);
				w_table[0] = 1;
				for (int i = 1; i < h/2; i++) {
					w_table[i] = barrett.multiply(w_table[i-1], wn);
				}
				w_tables.push_back(w_table);
			}
			
			// SIMD优化的蝶形运算
			// 处理2个元素为一组
			for (; k + 2 <= h/2; k += 2) {
				ULL w0 = w_table[k];

	\begin{lstlisting}[language=C++, caption=固定内存优化实现]
		void NTT_GPU_Level2_PinnedMemory(const LL *a, const LL *b, LL *ab, int n, LL p) {
			// 使用固定内存加速传输
			LL *h_A, *h_B;
			cudaMallocHost(&h_A, len * sizeof(LL));
			cudaMallocHost(&h_B, len * sizeof(LL));
			// 预先计算并传输所有旋转因子
			std::vector<std::vector<LL>> all_w_tables_forward;
			std::vector<std::vector<LL>> all_w_tables_inverse;
			
			for (int h = 2; h <= len; h <<= 1) {
				std::vector<LL> w_table(h/2);
				LL wn = qpowll(3, (p - 1) / h, p);
				w_table[0] = 1;
				for (int i = 1; i < h/2; i++) {
					w_table[i] = (w_table[i-1] * wn) % p;
				}
				all_w_tables_forward.push_back(w_table);
			}
			
			// 一次性传输所有数据
			cudaMemcpy(d_A, h_A, len * sizeof(LL), cudaMemcpyHostToDevice);
			cudaMemcpy(d_B, h_B, len * sizeof(LL), cudaMemcpyHostToDevice);
		}
	\end{lstlisting}
	
	\paragraph{Level 3: 流并行优化}
	CUDA流允许并发执行多个操作：

		void NTT_GPU_Level3_Streams(const LL *a, const LL *b, LL *ab, int n, LL p) {
			// 创建2个流分别处理A和B的NTT
			cudaStream_t streamA, streamB;
			cudaStreamCreate(&streamA);
			cudaStreamCreate(&streamB);
			// 异步传输
			cudaMemcpyAsync(d_A, h_A, len * sizeof(LL), cudaMemcpyHostToDevice, streamA);
			cudaMemcpyAsync(d_B, h_B, len * sizeof(LL), cudaMemcpyHostToDevice, streamB);
			
			// 异步执行位逆序
			bit_reverse_kernel<<<gridSize, blockSize, 0, streamA>>>(d_A, len);
			bit_reverse_kernel<<<gridSize, blockSize, 0, streamB>>>(d_B, len);
			
			// 正向NTT（并行处理A和B）
			for (int h = 2; h <= len; h <<= 1) {
				ntt_basic_kernel<<<butterflyGrid, butterflyBlock, 0, streamA>>>(
				d_A, len, h, d_w_table_A, p);
				ntt_basic_kernel<<<butterflyGrid, butterflyBlock, 0, streamB>>>(
				d_B, len, h, d_w_table_B, p);
			}
			
			// 等待两个流完成
			cudaStreamSynchronize(streamA);
			cudaStreamSynchronize(streamB);
		}
	\end{lstlisting}

	\subsection{性能瓶颈分析}
	通过实验数据和性能分析工具，我们识别了各种并行方案的主要瓶颈：
	\subsubsection{CPU并行瓶颈}
	\begin{itemize}
		\item \textbf{内存带宽限制}：NTT的蝶形运算存在大量不规则内存访问，缓存命中率较低
		\item \textbf{同步开销}：多线程间的barrier同步成为性能瓶颈，特别是在线程数较多时
		\item \textbf{负载不均衡}：NTT不同层级的计算量差异导致线程空闲
	\end{itemize}
	\subsubsection{GPU并行瓶颈}
	\begin{itemize}
		\item \textbf{启动开销}：小规模问题下kernel启动开销占主导
		\item \textbf{内存访问模式}：位逆序操作导致非合并内存访问
		\item \textbf{线程分歧}：条件分支导致warp内线程执行路径不一致
		\item \textbf{计算密度不足}：相对于内存操作，算术运算较少
	\end{itemize}
	\subsubsection{异构计算瓶颈}
	\begin{itemize}
		\item \textbf{通信开销}：CPU-GPU数据传输成为主要开销
		\item \textbf{任务粒度}：过细的任务划分增加调度开销
		\item \textbf{同步等待}：CPU和GPU计算速度不匹配导致等待
	\end{itemize}

		\item \textbf{性能建模}：学会了使用Amdahl定律、Roofline模型等工具分析性能上限。
		
		\item \textbf{权衡}：理解了性能、功耗、开发复杂度之间的权衡关系。
	\end{enumerate}
	\subsection{实践经验}
	\subsubsection{调试技巧}
	\begin{enumerate}
		\item \textbf{并行调试困难}：并行程序的非确定性使得bug难以重现，学会了使用各种调试工具和技巧。
		\item \textbf{性能分析工具}：掌握了VTune等性能分析工具的使用。
		
		\item \textbf{正确性验证}：建立完善的测试框架，确保优化不影响正确性。
	\end{enumerate}
	\subsubsection{工程实践}
	\begin{enumerate}
		\item \textbf{代码组织}：学会了如何组织大型并行程序，保持代码的可读性和可维护性。
		\item \textbf{跨平台开发}：处理不同硬件平台的差异，编写可移植的代码。
		
		\item \textbf{性能调优流程}：建立了系统的性能优化流程：分析瓶颈→设计方案→实现验证→迭代改进。
	\end{enumerate}
	\subsection{反思与展望}
	\subsubsection{不足之处}
	\begin{enumerate}
		\item \textbf{理论深度}：虽然掌握了各种并行技术的使用，但其实对根本原理、底层原理的理解还不够深入。
		\item \textbf{大规模系统}：实验主要在单机或小规模集群上进行，缺乏大规模分布式系统的实践经验。
		
		\item \textbf{新技术探索}：对一些新兴的并行技术（如FPGA、神经网络加速器）了解不足。

	\subsubsection{未来学习方向}
	\begin{enumerate}
		\item \textbf{深入硬件架构}：进一步学习计算机体系结构，理解硬件设计对并行性能的影响。

	\section{结论}
	本学期的并行程序设计课程让我全面掌握了现代并行计算的核心技术。通过NTT多项式乘法这一案例的深入研究，我不仅学会了各种并行编程技术的使用，更重要的是理解了并行优化的思维方式和方法论。
	期末综合实验通过设计三条不同的优化路线，展示了如何综合运用多种并行技术实现最优性能。实验结果表明：

		\item 算法优化和系统优化同等重要
		\item 性能优化需要全栈思维和系统方法
	\end{itemize}
	这门课程不仅提升了我的编程能力，更培养了我的系统思维和工程素养。
	并行计算作为支撑现代信息技术发展的基础，其重要性将随着数据规模的增长和应用需求的提升而不断凸显。掌握并行编程技术，不仅是技术发展的要求，更是时代赋予我们的使命。

void fRead(int *a, int *b, int *n, int *p, int input_id){
    std::string str1 = "nttdata/";
    std::string str2 = std::to_string(input_id);
    std::string strin = str1 + str2 + ".in";
    char data_path[strin.size() + 1];
    std::copy(strin.begin(), strin.end(), data_path);
    data_path[strin.size()] = '\0';
    std::ifstream fin;
    fin.open(data_path, std::ios::in);
    fin>>*n>>*p;
    for (int i = 0; i < *n; i++){
        fin>>a[i];
    }
    for (int i = 0; i < *n; i++){
        fin>>b[i];
    }
}

    std::string str2 = std::to_string(input_id);
    std::string strout = str1 + str2 + ".out";

    data_path[strout.size()] = '\0';
    std::ifstream fin;
    fin.open(data_path, std::ios::in);
    for (int i = 0; i < n * 2 - 1; i++){
        int x;
        fin>>x;
        if(x != ab[i]){
            std::cout<<"多项式乘法结果错误"<<std::endl;
            return;

int qpow(int a,int b,int p){//快速幂
  int res = 1;
  while (b) {
    if (b & 1) res = 1LL * res * a % p;
    a = 1LL * a * a % p;
    b >>= 1;
    }
  return res;
}

// x86 SIMD: 128-bit vectorized modmul for 4 int32_t at a time
static inline __m128i x86_mod_mul(__m128i a, __m128i b, __m128i p) {
    alignas(16) int32_t a_array[4], b_array[4], p_array[4], result[4];
    _mm_store_si128((__m128i*)a_array, a);
    _mm_store_si128((__m128i*)b_array, b);
    _mm_store_si128((__m128i*)p_array, p);
    for (int i = 0; i < 4; i++) {
        result[i] = (int64_t(a_array[i]) * b_array[i]) % p_array[i];
        if (result[i] < 0) result[i] += p_array[i];
    }
    return _mm_load_si128((__m128i*)result);
}

static inline __m128i x86_mod(__m128i a, __m128i p) {
    alignas(16) int32_t a_array[4], p_array[4], result[4];
    _mm_store_si128((__m128i*)a_array, a);
    _mm_store_si128((__m128i*)p_array, p);
    for (int i = 0; i < 4; i++) {
        result[i] = a_array[i] % p_array[i];
        if (result[i] < 0) result[i] += p_array[i];
    }
    return _mm_load_si128((__m128i*)result);
}

struct MontgomeryParams {
    uint32_t mod;
    uint32_t n_prime;
    uint32_t r_squared;
};

static inline MontgomeryParams init_montgomery_params(uint32_t modulus) {
    MontgomeryParams params;
    params.mod = modulus;
    uint32_t x = 1;
    for (int i = 0; i < 5; i++) x = x * (2 - modulus * x);
    params.n_prime = -x;
    uint64_t r_squared = (1ULL << 32) % modulus;
    r_squared = (r_squared * r_squared) % modulus;
    params.r_squared = (uint32_t)r_squared;
    return params;
}

static inline uint32_t montgomery_reduce_scalar(uint64_t T, const MontgomeryParams* params) {
    uint32_t m = (uint32_t)T * params->n_prime;
    uint64_t t = (T + (uint64_t)m * params->mod) >> 32;
    return (t >= params->mod) ? (t - params->mod) : t;
}

static inline uint32_t to_montgomery_scalar(uint32_t a, const MontgomeryParams* params) {
    return montgomery_reduce_scalar((uint64_t)a * params->r_squared, params);
}

static inline uint32_t from_montgomery_scalar(uint32_t a, const MontgomeryParams* params) {
    return montgomery_reduce_scalar((uint64_t)a, params);
}

static inline __m128i montgomery_multiply_x86(__m128i a, __m128i b, const MontgomeryParams* params) {
    alignas(16) uint32_t a_array[4], b_array[4], result[4];
    uint32_t n_prime = params->n_prime;
    uint32_t mod = params->mod;
    _mm_store_si128((__m128i*)a_array, a);
    _mm_store_si128((__m128i*)b_array, b);
    for (int i = 0; i < 4; ++i) {
        uint64_t T = uint64_t(a_array[i]) * b_array[i];
        uint32_t m = (uint32_t)T * n_prime;
        uint64_t t = (T + (uint64_t)m * mod) >> 32;
        result[i] = (t >= mod) ? (t - mod) : t;
    }
    return _mm_load_si128((__m128i*)result);
}

static void prepare_w_table(uint32_t* W, int len, uint32_t wn, uint32_t p, const MontgomeryParams* params) {
    W[0] = to_montgomery_scalar(1, params);
    for (int i = 1; i < len; i++) {
        W[i] = montgomery_reduce_scalar((uint64_t)W[i-1] * wn, params);
    }
}

uint32_t montgomery_pow(uint32_t a, int b, const MontgomeryParams* params) {
    uint32_t res = to_montgomery_scalar(1, params);
    while (b) {
        if (b & 1) res = montgomery_reduce_scalar((uint64_t)res * a, params);
        a = montgomery_reduce_scalar((uint64_t)a * a, params);
        b >>= 1;
    }
    return res;
}

void poly_multiply(int *a, int *b, int *ab, int n, int p){
    for(int i = 0; i < n; ++i){
        for(int j = 0; j < n; ++j){
            ab[i+j]=(1LL * a[i] * b[j] % p + ab[i+j]) % p;
        }
    }
}

void poly_multiply_simd(int *a, int *b, int *ab, int n, int p) {
    for (int i = 0; i < 2 * n - 1; ++i) ab[i] = 0;
    for (int i = 0; i < n; ++i) {
        int a_val = a[i];
        __m128i a_vec = _mm_set1_epi32(a_val);
        for (int j = 0; j + 3 < n; j += 4) {
            __m128i b_vec = _mm_loadu_si128((__m128i const*)(&b[j]));
            __m128i ab_vec = _mm_loadu_si128((__m128i const*)(&ab[i + j]));
            __m128i p_vec = _mm_set1_epi32(p);
            __m128i mul = x86_mod_mul(a_vec, b_vec, p_vec);
            ab_vec = _mm_add_epi32(ab_vec, mul);
            ab_vec = x86_mod(ab_vec, p_vec);
            _mm_storeu_si128((__m128i*)(&ab[i + j]), ab_vec);
        }
        for (int j = (n & ~3); j < n; ++j) {
            ab[i + j] = (1LL * a[i] * b[j] % p + ab[i + j]) % p;
        }
    }
}

const int g=3;

void NTT_recursion(int *a,int len,int op,int g,int p){
    if (len == 1) return;
    int half = len / 2;
    int *A1 = (int*)alloca(sizeof(int) * half);
    int *A2 = (int*)alloca(sizeof(int) * half);

    for (int i = 0; i < half; ++i) {
        A1[i] = a[i * 2];
        A2[i] = a[i * 2 + 1];
    }

    NTT_recursion(A1, half, op, g, p);
    NTT_recursion(A2, half, op, g, p);

    int wn = qpow((op == 1 ? g : qpow(g, p - 2, p)), (p - 1) / len, p);
    int w = 1;

    for (int i = 0; i < half; ++i) {
        int t = 1LL * A2[i] * w % p;
        a[i] = (A1[i] + t) % p;
        a[i + half] = (A1[i] - t + p) % p;
        w = 1LL * w * wn % p;
    }
}

void NTT_recursion_multiply(int *a, int *b, int *ab, int n, int p){
    int N = 2 * n - 1;
    int len = 1;
    int ni;
    while (len < N) len <<= 1;
    ni=qpow(len,p-2,p);
    int *A = new int[len]();
    int *B = new int[len]();
    for (int i = 0; i < len; ++i) {
        A[i] = (i < n) ? a[i] : 0;
        B[i] = (i < n) ? b[i] : 0;
    }
    NTT_recursion(A, len, 1, g, p);
    NTT_recursion(B, len, 1, g, p);
    for (int i = 0; i < len; ++i) {
        A[i] = (1LL * A[i] * B[i]) % p;
    }
    NTT_recursion(A, len, -1, g, p);
    for (int i = 0; i < N; ++i) {
        ab[i] = (1LL * A[i] * ni) % p;
    }
    delete[] A;
    delete[] B;
}

void NTT_recursion_mont(uint32_t *a, int len, int op, uint32_t g, uint32_t p, const MontgomeryParams *params) {
    if (len == 1) return;
    int half = len / 2;
    uint32_t *A1 = (uint32_t*)alloca(sizeof(uint32_t) * half);
    uint32_t *A2 = (uint32_t*)alloca(sizeof(uint32_t) * half);

    for (int i = 0; i < half; ++i) {
        A1[i] = a[i * 2];
        A2[i] = a[i * 2 + 1];
    }

    NTT_recursion_mont(A1, half, op, g, p, params);
    NTT_recursion_mont(A2, half, op, g, p, params);

    uint32_t root = (op == 1 ? g : qpow(g, p - 2, p));
    uint32_t root_mont = to_montgomery_scalar(root, params);
    uint32_t wn = montgomery_pow(root_mont, (p - 1) / len, params);
    uint32_t w = to_montgomery_scalar(1, params);

    for (int i = 0; i < half; ++i) {
        uint32_t t = montgomery_reduce_scalar((uint64_t)A2[i] * w, params);
        uint32_t add = A1[i] + t;
        if (add >= p) add -= p;
        a[i] = add;
        uint32_t sub = A1[i] + p - t;
        if (sub >= p) sub -= p;
        a[i + half] = sub;
        w = montgomery_reduce_scalar((uint64_t)w * wn, params);
    }
}

void NTT_recursion_multiply_SIMD(int *a, int *b, int *ab, int n, int p, int g=3) {
    const int N = 2 * n - 1;
    int len = 1;
    while (len < N) len <<= 1;
    int ni = qpow(len, p - 2, p);

    MontgomeryParams params = init_montgomery_params(p);

    uint32_t *A = (uint32_t*)ALIGNED_ALLOC(len * sizeof(uint32_t), 16);
    uint32_t *B = (uint32_t*)ALIGNED_ALLOC(len * sizeof(uint32_t), 16);
    for (int i = 0; i < len; ++i) {
        A[i] = (i < n) ? to_montgomery_scalar(a[i], &params) : 0;
        B[i] = (i < n) ? to_montgomery_scalar(b[i], &params) : 0;
    }

    NTT_recursion_mont(A, len, 1, g, p, &params);
    NTT_recursion_mont(B, len, 1, g, p, &params);

    for (int i = 0; i < len; ++i) {
        A[i] = montgomery_reduce_scalar((uint64_t)A[i] * B[i], &params);
    }

    NTT_recursion_mont(A, len, -1, g, p, &params);

    for (int i = 0; i < N; ++i) {
        ab[i] = (uint64_t)from_montgomery_scalar(A[i], &params) * ni % p;
    }

    ALIGNED_FREE(A);
    ALIGNED_FREE(B);
}

void NTT_Iteration(int *a, int len, int op, int g, int p) {
    for (int i = 1, j = 0; i < len - 1; i++) {
        for (int k = len >> 1; (j ^= k) < k; k >>= 1);
        if (i < j) {
            int tmp = a[i];
            a[i] = a[j];
            a[j] = tmp;
        }
    }
    for (int h = 2; h <= len; h <<= 1) {
        int wn = qpow((op == 1 ? g : qpow(g, p - 2, p)), (p - 1) / h, p);
        for (int j = 0; j < len; j += h) {
            int w = 1;
            for (int k = j; k < j + h/2; k++) {
                int t = 1LL * a[k + h/2] * w % p;
                a[k + h/2] = (a[k] - t + p) % p;
                a[k] = (a[k] + t) % p;
                w = 1LL * w * wn % p;
            }
        }
    }
}

void NTT_Iteration_multiply(int *a, int *b, int *ab, int n, int p) {
    int N = 2 * n - 1;
    int len = 1;
    while (len < N) len <<= 1;
    int ni = qpow(len, p-2, p);
    int *A = new int[len]();
    int *B = new int[len]();
    for (int i = 0; i < len; i++) {
        A[i] = (i < n) ? a[i] : 0;
        B[i] = (i < n) ? b[i] : 0;
    }
    NTT_Iteration(A, len, 1, g, p);
    NTT_Iteration(B, len, 1, g, p);
    for (int i = 0; i < len; i++) {
        A[i] = (1LL * A[i] * B[i]) % p;
    }
    NTT_Iteration(A, len, -1, g, p);
    for (int i = 0; i < N; i++) {
        ab[i] = (1LL * A[i] * ni) % p;
    }
    delete[] A;
    delete[] B;
}

void NTT_Iteration_SIMD(uint32_t* a, int len, int op, uint32_t g, uint32_t p, const MontgomeryParams* params) {
    for (int i = 1, j = 0; i < len-1; i++) {
        for (int k = len >> 1; (j ^= k) < k; k >>= 1);
        if (i < j) { uint32_t tmp = a[i]; a[i] = a[j]; a[j] = tmp; }
    }
    uint32_t* W = (uint32_t*)ALIGNED_ALLOC(len * sizeof(uint32_t), 16);
    for (int h = 2; h <= len; h <<= 1) {
        uint32_t root = (op == 1 ? g : qpow(g, p-2, p));
        uint32_t root_mont = to_montgomery_scalar(root, params);
        uint32_t wn = montgomery_pow(root_mont, (p-1)/h, params);
        prepare_w_table(W, h/2, wn, p, params);
        for (int j = 0; j < len; j += h) {
            int k = 0;
            for (; k + 4 <= h/2; k += 4) {
                __m128i vA = _mm_loadu_si128((__m128i const*)(a + j + k));
                __m128i vB = _mm_loadu_si128((__m128i const*)(a + j + k + h/2));
                __m128i vW = _mm_loadu_si128((__m128i const*)(W + k));
                __m128i vT = montgomery_multiply_x86(vB, vW, params);
                __m128i vSum = _mm_add_epi32(vA, vT);
                __m128i vDiff = _mm_sub_epi32(vA, vT);
                alignas(16) uint32_t sum_arr[4], diff_arr[4], t_arr[4], u_arr[4];
                _mm_store_si128((__m128i*)sum_arr, vSum);
                _mm_store_si128((__m128i*)diff_arr, vDiff);
                _mm_store_si128((__m128i*)t_arr, vT);
                _mm_store_si128((__m128i*)u_arr, vA);
                for (int ii = 0; ii < 4; ++ii) {
                    if (sum_arr[ii] >= p) sum_arr[ii] -= p;
                    if (t_arr[ii] > u_arr[ii]) diff_arr[ii] += p;
                }
                _mm_storeu_si128((__m128i*)(a + j + k), _mm_load_si128((__m128i*)sum_arr));
                _mm_storeu_si128((__m128i*)(a + j + k + h/2), _mm_load_si128((__m128i*)diff_arr));
            }
            for (; k < h/2; ++k) {
                uint32_t t = montgomery_reduce_scalar((uint64_t)a[j+k+h/2] * W[k], params);
                uint32_t u = a[j+k];
                uint32_t add = u + t;
                if (add >= p) add -= p;
                a[j+k] = add;
                uint32_t sub = u + p - t;
                if (sub >= p) sub -= p;
                a[j+k+h/2] = sub;
            }
        }
    }
    ALIGNED_FREE(W);
}

void NTT_Iteration_multiply_SIMD(int* a, int* b, int* ab, int n, int p) {
    const int N = 2*n-1;
    int len = 1;
    while (len < N) len <<= 1;
    int ni = qpow(len, p-2, p);

    uint32_t* A = (uint32_t*)ALIGNED_ALLOC(len * sizeof(uint32_t), 16);
    uint32_t* B = (uint32_t*)ALIGNED_ALLOC(len * sizeof(uint32_t), 16);

    MontgomeryParams params = init_montgomery_params(p);

    for (int i = 0; i < len; ++i) {
        A[i] = (i < n) ? to_montgomery_scalar(a[i], &params) : 0;
        B[i] = (i < n) ? to_montgomery_scalar(b[i], &params) : 0;
    }

    NTT_Iteration_SIMD(A, len, 1, 3, p, &params);
    NTT_Iteration_SIMD(B, len, 1, 3, p, &params);

    for (int i = 0; i < len; ++i) {
        A[i] = montgomery_reduce_scalar((uint64_t)A[i] * B[i], &params);
    }

    NTT_Iteration_SIMD(A, len, -1, 3, p, &params);

    for (int i = 0; i < N; ++i) {
        ab[i] = (uint64_t)from_montgomery_scalar(A[i], &params) * ni % p;
    }

    ALIGNED_FREE(A);
    ALIGNED_FREE(B);
}

int a[300000], b[300000], ab[300000];
int main(int argc, char *argv[])
{
    int test_begin = 0;
    int test_end = 0;
    for(int i = test_begin; i <= test_end; ++i){
        long double ans = 0;
        int n_, p_;
        fRead(a, b, &n_, &p_, i);
        memset(ab,0,sizeof(ab));
        auto Start = std::chrono::high_resolution_clock::now();
        poly_multiply(a, b, ab, n_, p_);
        poly_multiply_simd(a, b, ab, n_, p_);
        NTT_recursion_multiply(a, b, ab, n_, p_);
        NTT_recursion_multiply_SIMD(a, b, ab, n_, p_);
        NTT_Iteration_multiply(a, b, ab, n_, p_);
        NTT_Iteration_multiply_SIMD(a, b, ab, n_, p_);
        auto End = std::chrono::high_resolution_clock::now();
        std::chrono::duration<double,std::ratio<1,1000>>elapsed = End - Start;
        ans += elapsed.count();
        fCheck(ab, n_, i);
        std::cout<<"average latency for n = "<<n_<<" p = "<<p_<<" : "<<ans<<" (us) "<<std::endl;
        fWrite(ab, n_, i);
    }
    return 0;
}
